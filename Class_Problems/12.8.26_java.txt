    }

    }

        System.out.println("this is a traiangle");

    void area(double a, int b){
        System.out.println("this is a circle");

    }
}

abstract class employee{
    private String name;
    private float salary;
    void get_name(String name){
        this.name=name;
    }
    void get_salary(float salary){
        this.salary=salary;
    }
    String set_name(){
        return name;
    }
    float set_salary(){
        return salary;
    }
    abstract float bonus();
    abstract void display(); 

}
class developers extends employee{
    @Override
    float bonus(){
        float bal=set_salary();
        float bonus=0.05f;
        float b=(float)(bonus*bal);
        return b;
    }
    @Override
    void display(){
        System.out.println("name "+set_name());
        System.out.println("salary "+set_salary());
        System.out.println("bonus is "+ bonus());
        System.out.println("total salary "+(set_salary()+bonus()));
    }
}

    public static void main(String args[]){
        Scanner sc=new Scanner(System.in);
        int a=sc.nextInt();
        int b=sc.nextInt();
        try {
            if(b!=0){
                System.out.println(0);
            }
            else{
                int c=a/b;
            }
        } catch (ArithmeticException e) {
            System.out.println("number is not divisible by zero");
        }
        int arr[]=new int[5];
        try {
            System.out.println(arr[7]);
        } catch (ArrayIndexOutOfBoundsException e) {
            System.out.println("can't able to find the element");
        }
        String s=null;
        try {
            System.out.println(s.charAt(3));
        } catch (NullPointerException e) {
            System.out.println("there is no element");
        }
    }
}

class bank{
    int balance;
    bank(int balance){
        this.balance=balance;
    }
    void withdraw(int amt){
        if(amt<=balance){
            balance-=amt;
            System.out.println("the remaining balance is "+balance);
        }
        else{
            throw new RuntimeException("invalid balance.");
        }
    }
}
class second_java{
    public static void main(String args[]){
        bank b=new bank(400);
        b.withdraw(4000);
    }
}
