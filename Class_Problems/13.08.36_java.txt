            balance-=amt;
            System.out.println("remaining balance "+balance);
        }

            throw new InsufficientBalanceException("you dont have eough balance");

   }

class InvalidMarkException extends RuntimeException{
    public InvalidMarkException(String e){
        super(e);
    }
}
class ArrayLengthException extends Exception{
    public ArrayLengthException(String e){
        super(e);
    }
}
class second_java {
    public static void main(String args[]){
        Scanner sc=new Scanner(System.in);
        System.out.println("enter the number of subjects ");
        int n=sc.nextInt();
        try {//0< ,100> array lr -5 (compile time)\
            if(n!=5){
                throw new ArrayLengthException("enter only 5 sub");
            }
        } catch (ArrayLengthException e){
            System.out.println(e.getMessage());
            return;
        }
        int arr[]=new int[n];
        int avg=0;
        for (int i = 0; i <n; i++) {            
                arr[i]=sc.nextInt();
        }
            for(int i=0;i<n;i++){
            try {
                if(arr[i]<0){
                    throw new InsufficientMarkException(("enter crt mark"));
                }
                if(arr[i]>100){
                    throw new InvalidMarkException(("enter your original mark"));
                }
                avg+=arr[i];
            } catch (InsufficientMarkException e) {
                System.out.println(e.getMessage());
                return;
            }
            catch(InvalidMarkException e){
                System.out.println(e.getMessage());
                return;
            }
            }
        System.out.println(avg/n);   
    }
}

import java.util.*;
class second_java{
    public static void main(String args[]){
        ArrayList <Integer> list=new ArrayList<>();
        Scanner sc=new Scanner(System.in);
        int a;
        while (true) { 
            a=sc.nextInt();
            if(a==-1){
                break;
            }
            else{
                list.add(a);
            }
        }
        for(int i=0;i<=list.size()-2;i++){
            for(int j=0;j<=list.size()-2-i;j++){
                if(list.get(j)>list.get(j+1)){
                    int t=list.get(j);         // to get a value in array list 
                    list.set(j,list.get(j+1));  // to set the value in array list 1 st index then value
                    list.set(j+1,t);
                }
            }
        }
        System.out.println("sorted Array "+list);
    }
}
